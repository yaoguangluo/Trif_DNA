 private int partition(float[] array, int leftPoint, int rightPoint) {
		    	float x = array[leftPoint] <= array[rightPoint] ? array[leftPoint] : array[rightPoint];
		        int leftPointReflection = leftPoint;
		        while (leftPointReflection < rightPoint) {
		            while (!(array[leftPointReflection] > x || leftPointReflection++ >= rightPoint)) {
		            }
		            while (array[rightPoint--] > x) {
		            }
		            if (leftPointReflection < ++rightPoint) {
		            	float temp = array[rightPoint];
		                array[rightPoint] = array[leftPointReflection];
		                array[leftPointReflection] = temp;
		            }
		        }
		        array[leftPoint] = array[rightPoint];
		        array[rightPoint] = x;
		        return rightPoint;
		    }

				    l = leftPoint;

				        }

				            array[l] = temp;

				        node.R = pos - 1;
				        end.next = node;
				        end= end.next;

				        node.R = rightPoint;

				        end= end.next;

			    	int i = leftPoint;
			        for (int j = i + 1; j < leftPoint + c; j = i++) {
			        	while (j > leftPoint) {
			        		if (array[j] < array[--j]) {

			        			array[j] = temp;
			        		}
			        	}
			        }
			        }
			        cur = cur.next;

			        	float x = array[leftPoint] <= array[rightPoint] ? array[leftPoint] : array[rightPoint];
				        int leftPointReflection = leftPoint;
				        int r = rightPoint;
				        while (leftPointReflection < r) {
				            while (!(array[leftPointReflection] > x || leftPointReflection++ >= r)) {
				            }
				            while (array[r--] > x) {
				            }
				            if (leftPointReflection < ++r) {
				            	float temp = array[r];
				                array[r] = array[leftPointReflection];
				                array[leftPointReflection] = temp;
				            }
				        }
				        array[leftPoint] = array[r];
				        array[r] = x;   
			            if (leftPoint < r - 1) {
			                processDouble(array, leftPoint, r - 1, deep + 1);
			            }
			            if (r + 1 < rightPoint) {
			                processDouble(array, r + 1, rightPoint, deep + 1);
			            }
			            return;
			        }
			        int i = leftPoint;
			        for (int j = i + 1; j < leftPoint + c; j = i++) {

			        			float temp = array[j + 1];

			        		}
			        	}
			        }
		    }
